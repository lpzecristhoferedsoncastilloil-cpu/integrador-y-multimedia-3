# 6. LOGIN COMPLETO

## frontend/src/pages/Login.jsx

```jsx
import { useState } from 'react'
import { useNavigate } from 'react-router-dom'
import Swal from 'sweetalert2'
import api from '../services/api'

export default function Login() {

  const navigate = useNavigate()

  const [correo, setCorreo] = useState('')
  const [password, setPassword] = useState('')

  const iniciarSesion = async (e) => {

    e.preventDefault()

    try {

      const respuesta = await api.post('token/', {
        correo,
        password
      })

      localStorage.setItem('token', respuesta.data.access)
      localStorage.setItem('rol', respuesta.data.rol)

      Swal.fire({
        icon: 'success',
        title: 'Bienvenido',
        text: 'Inicio de sesión correcto'
      })

      if (respuesta.data.rol === 'psicologo') {
        navigate('/dashboard')
      } else {
        navigate('/inicio-paciente')
      }

    } catch (error) {

      Swal.fire({
        icon: 'error',
        title: 'Error',
        text: 'Correo o contraseña incorrectos'
      })
    }
  }

  return (

    <div className='min-h-screen bg-gradient-to-r from-cyan-500 to-blue-700 flex items-center justify-center'>

      <form
        onSubmit={iniciarSesion}
        className='bg-white p-10 rounded-3xl shadow-2xl w-[400px]'
      >

        <h1 className='text-4xl font-bold text-center mb-8'>
          NeuroGym
        </h1>

        <input
          type='email'
          placeholder='Correo electrónico'
          value={correo}
          onChange={(e) => setCorreo(e.target.value)}
          className='w-full p-4 border rounded-xl mb-5'
        />

        <input
          type='password'
          placeholder='Contraseña'
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          className='w-full p-4 border rounded-xl mb-5'
        />

        <button
          className='w-full bg-blue-600 text-white p-4 rounded-xl hover:bg-blue-800 transition'
        >
          Iniciar Sesión
        </button>

      </form>

    </div>
  )
}
```
